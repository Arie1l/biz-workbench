配置步骤（免费，约3分钟）
1
打开 supabase.com，用 GitHub 账号登录（点 Sign in → Continue with GitHub）
2
登录后点 New Project，名字随便填（如 my-workbench），密码自动生成不用管，地区选 Singapore，点 Create
3
等项目建好后（约1-2分钟），点左侧 SQL Editor → New query，把下面的 SQL 复制粘贴进去，点 Run
CREATE TABLE IF NOT EXISTS orders ( id TEXT PRIMARY KEY, order_number TEXT, project_name TEXT, customer TEXT, salesperson TEXT, proposal_number TEXT DEFAULT '', contract_amount NUMERIC DEFAULT 0, created_at TEXT DEFAULT '', status TEXT DEFAULT 'active', remark TEXT DEFAULT '', steps JSONB DEFAULT '{}'::jsonb ); CREATE TABLE IF NOT EXISTS tasks ( id TEXT PRIMARY KEY, title TEXT, description TEXT DEFAULT '', date TEXT, priority TEXT DEFAULT 'medium', completed BOOLEAN DEFAULT FALSE, related_order TEXT DEFAULT '', salesperson TEXT DEFAULT '' ); ALTER TABLE orders ENABLE ROW LEVEL SECURITY; ALTER TABLE tasks ENABLE ROW LEVEL SECURITY; CREATE POLICY "all_orders" ON orders FOR ALL TO anon USING (true) WITH CHECK (true); CREATE POLICY "all_tasks" ON tasks FOR ALL TO anon USING (true) WITH CHECK (true);
4
点左侧 Project Settings（齿轮图标）→ API，找到 Project URL 和 anon public key，复制后填到下面
